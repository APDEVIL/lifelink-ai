// components/corridor/signal-log.tsx
// Police control room: live table of every signal the AI has flipped.
// Shows junction name, road-link, current state, and time.
// Pulses green/red on new events from Pusher.

"use client";

import { useEffect, useRef, useState } from "react";
import { Badge } from "@/components/ui/badge";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { cn } from "@/lib/utils";

export interface SignalLogEntry {
  signalId: string;
  junctionName: string;
  roadLinkId: string;
  roadLinkDescription?: string;
  state: "green" | "red";
  action?: "GREEN" | "RED";
  ambulanceEtaSeconds?: number;
  timestamp: string;
}

interface SignalLogProps {
  entries: SignalLogEntry[];
  className?: string;
  maxRows?: number;
}

export function SignalLog({ entries, className, maxRows = 50 }: SignalLogProps) {
  const scrollRef = useRef<HTMLDivElement>(null);
  const [flashIds, setFlashIds] = useState<Set<string>>(new Set());
  const prevCountRef = useRef(entries.length);

  // Auto-scroll to bottom on new entries
  useEffect(() => {
    const el = scrollRef.current;
    if (!el) return;
    el.scrollTop = el.scrollHeight;
  }, [entries.length]);

  // Flash new rows
  useEffect(() => {
    if (entries.length <= prevCountRef.current) return;
    const newEntries = entries.slice(prevCountRef.current);
    const newIds = new Set(newEntries.map((e) => `${e.signalId}-${e.timestamp}`));
    setFlashIds(newIds);
    const timer = setTimeout(() => setFlashIds(new Set()), 1200);
    prevCountRef.current = entries.length;
    return () => clearTimeout(timer);
  }, [entries]);

  const displayEntries = entries.slice(-maxRows).reverse();

  return (
    <Card className={cn("flex flex-col", className)}>
      <CardHeader className="pb-2 border-b border-slate-800">
        <div className="flex items-center justify-between">
          <CardTitle className="text-sm font-semibold text-slate-200 tracking-wide uppercase">
            Signal Interruption Log
          </CardTitle>
          <div className="flex items-center gap-2">
            <span className="h-2 w-2 rounded-full bg-emerald-400 animate-pulse" />
            <span className="text-xs text-slate-400">Live</span>
          </div>
        </div>
        <p className="text-xs text-slate-500 mt-0.5">
          AI-managed signal changes — {entries.length} total interventions
        </p>
      </CardHeader>

      <CardContent className="p-0 flex-1 overflow-hidden">
        {displayEntries.length === 0 ? (
          <div className="flex items-center justify-center h-32 text-slate-500 text-sm">
            No signal interventions yet
          </div>
        ) : (
          <div ref={scrollRef} className="overflow-y-auto max-h-[420px]">
            <table className="w-full text-xs">
              <thead className="sticky top-0 bg-slate-900 z-10">
                <tr className="text-slate-500 border-b border-slate-800">
                  <th className="text-left px-4 py-2 font-medium">Junction</th>
                  <th className="text-left px-4 py-2 font-medium hidden md:table-cell">Road Link</th>
                  <th className="text-center px-4 py-2 font-medium">State</th>
                  <th className="text-right px-4 py-2 font-medium hidden sm:table-cell">ETA</th>
                  <th className="text-right px-4 py-2 font-medium">Time</th>
                </tr>
              </thead>
              <tbody>
                {displayEntries.map((entry, i) => {
                  const key = `${entry.signalId}-${entry.timestamp}`;
                  const isFlashing = flashIds.has(key);
                  const isGreen = entry.state === "green";

                  return (
                    <tr
                      key={key}
                      className={cn(
                        "border-b border-slate-800/60 transition-colors duration-300",
                        i % 2 === 0 ? "bg-slate-900" : "bg-slate-900/60",
                        isFlashing && isGreen && "bg-emerald-950/60",
                        isFlashing && !isGreen && "bg-red-950/40"
                      )}
                    >
                      {/* Junction */}
                      <td className="px-4 py-2.5">
                        <span className="font-medium text-slate-200">{entry.junctionName}</span>
                        {entry.roadLinkDescription && (
                          <span className="block text-slate-500 text-[10px] mt-0.5 md:hidden">
                            {entry.roadLinkDescription}
                          </span>
                        )}
                      </td>

                      {/* Road Link (desktop) */}
                      <td className="px-4 py-2.5 text-slate-400 hidden md:table-cell">
                        <span className="font-mono text-[10px] bg-slate-800 px-1.5 py-0.5 rounded">
                          {entry.roadLinkId}
                        </span>
                        {entry.roadLinkDescription && (
                          <span className="ml-2 text-slate-500">{entry.roadLinkDescription}</span>
                        )}
                      </td>

                      {/* State badge */}
                      <td className="px-4 py-2.5 text-center">
                        <Badge
                          className={cn(
                            "text-[10px] font-bold px-2 py-0.5 border-0",
                            isGreen
                              ? "bg-emerald-500/20 text-emerald-400"
                              : "bg-red-500/20 text-red-400"
                          )}
                        >
                          {isGreen ? "● GREEN" : "● RED"}
                        </Badge>
                      </td>

                      {/* ETA */}
                      <td className="px-4 py-2.5 text-right text-slate-400 hidden sm:table-cell">
                        {entry.ambulanceEtaSeconds != null
                          ? `${Math.round(entry.ambulanceEtaSeconds)}s`
                          : "—"}
                      </td>

                      {/* Time */}
                      <td className="px-4 py-2.5 text-right text-slate-500 font-mono">
                        {formatTime(entry.timestamp)}
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        )}
      </CardContent>
    </Card>
  );
}

function formatTime(iso: string): string {
  try {
    return new Date(iso).toLocaleTimeString("en-IN", {
      hour: "2-digit",
      minute: "2-digit",
      second: "2-digit",
      hour12: false,
    });
  } catch {
    return "—";
  }
}
