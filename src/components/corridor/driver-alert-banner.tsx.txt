// components/corridor/driver-alert-banner.tsx
// Full-width urgent banner pushed to all drivers on the corridor route.
// Subscribes to Pusher "corridor-alerts" channel directly.
// Auto-dismisses after 30 seconds.

"use client";

import { useEffect, useState, useCallback } from "react";
import { usePusher } from "@/hooks/use-pusher";
import { cn } from "@/lib/utils";

interface DriverAlert {
  emergencyId: string;
  message: string;
  ambulanceLocation: { lat: number; lng: number };
  ambulanceHeading: string;
  etaSeconds: number;
  corridorRoads: string[];
  timestamp: string;
}

interface DriverAlertBannerProps {
  /** Pass emergencyId to also listen on the specific emergency channel */
  emergencyId?: string;
  className?: string;
}

export function DriverAlertBanner({ emergencyId, className }: DriverAlertBannerProps) {
  const [alert, setAlert] = useState<DriverAlert | null>(null);
  const [visible, setVisible] = useState(false);
  const [dismissTimer, setDismissTimer] = useState<ReturnType<typeof setTimeout> | null>(null);

  const handleAlert = useCallback((data: DriverAlert) => {
    setAlert(data);
    setVisible(true);

    // Clear any existing timer
    if (dismissTimer) clearTimeout(dismissTimer);

    // Auto-dismiss after 30 seconds
    const timer = setTimeout(() => {
      setVisible(false);
      setTimeout(() => setAlert(null), 500); // wait for fade-out
    }, 30_000);

    setDismissTimer(timer);
  }, [dismissTimer]);

  const handleClose = useCallback(() => {
    setVisible(false);
    if (dismissTimer) clearTimeout(dismissTimer);
    setTimeout(() => setAlert(null), 300);
  }, [dismissTimer]);

  // Subscribe to city-wide driver alerts
  usePusher("corridor-alerts", [
    { event: "driver-alert", handler: handleAlert },
  ]);

  // Also subscribe to emergency-specific channel if provided
  usePusher(
    emergencyId ? `emergency-${emergencyId}` : null,
    [{ event: "driver-alert", handler: handleAlert }],
    !!emergencyId
  );

  useEffect(() => {
    return () => {
      if (dismissTimer) clearTimeout(dismissTimer);
    };
  }, [dismissTimer]);

  if (!alert) return null;

  const roadsText = alert.corridorRoads?.join(", ") ?? "current route";
  const etaText = alert.etaSeconds > 0 ? `${Math.round(alert.etaSeconds)}s away` : "approaching";

  return (
    <div
      className={cn(
        "fixed top-0 left-0 right-0 z-50 transition-transform duration-300",
        visible ? "translate-y-0" : "-translate-y-full",
        className
      )}
      role="alert"
      aria-live="assertive"
    >
      {/* Flashing red top bar */}
      <div className="h-1 bg-red-500 animate-pulse" />

      <div className="bg-red-600 text-white shadow-2xl">
        <div className="max-w-5xl mx-auto px-4 py-3 flex items-start gap-4">
          {/* Siren icon */}
          <div className="flex-shrink-0 mt-0.5">
            <span className="text-2xl select-none" aria-hidden>🚨</span>
          </div>

          {/* Message */}
          <div className="flex-1 min-w-0">
            <p className="font-bold text-base leading-tight">
              AMBULANCE — CLEAR THE ROAD
            </p>
            <p className="text-red-100 text-sm mt-0.5">
              {alert.message}
            </p>
            <div className="flex flex-wrap gap-x-4 gap-y-1 mt-1.5 text-xs text-red-200">
              <span>📍 Corridor: {roadsText}</span>
              <span>⏱ {etaText}</span>
            </div>
          </div>

          {/* ETA badge */}
          <div className="flex-shrink-0 text-center hidden sm:block">
            <div className="bg-white/20 rounded-lg px-3 py-1.5">
              <p className="text-xl font-bold leading-none">
                {alert.etaSeconds > 0 ? Math.round(alert.etaSeconds) : "—"}
              </p>
              <p className="text-[10px] text-red-200 mt-0.5">seconds</p>
            </div>
          </div>

          {/* Dismiss */}
          <button
            onClick={handleClose}
            className="flex-shrink-0 text-red-200 hover:text-white transition-colors p-1 rounded"
            aria-label="Dismiss alert"
          >
            <svg className="w-5 h-5" fill="none" viewBox="0 0 24 24" stroke="currentColor">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
            </svg>
          </button>
        </div>
      </div>

      {/* Auto-dismiss progress bar */}
      {visible && (
        <div className="h-0.5 bg-red-800">
          <div
            className="h-full bg-white/50"
            style={{
              animation: "shrink 30s linear forwards",
            }}
          />
        </div>
      )}

      <style>{`
        @keyframes shrink {
          from { width: 100%; }
          to   { width: 0%; }
        }
      `}</style>
    </div>
  );
}
